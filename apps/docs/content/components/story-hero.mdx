## Usage

```tsx
import { StoryHero } from '@brand-studio/ui';
```

```tsx
<StoryHero
  title="Make room for the first sip."
  description="…"
  asset={cup}
  action={<ActionLink href="/menu">See the menu</ActionLink>}
/>
```
